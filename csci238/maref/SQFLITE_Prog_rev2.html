SQFLITE

Program Summary

This Flutter application demonstrates how to integrate the Sqflite package to manage data locally on a device. It lets the user add, edit, delete, and view car records—each with a make, model, year, and color. Data is stored persistently in a SQLite database using the CarDatabase helper class. The app’s interface features a simple form at the top for input and a scrollable ListView below that displays all saved cars. Each entry includes buttons to edit or delete a record. This project teaches CRUD (Create, Read, Update, Delete) operations, asynchronous database handling, and dynamic UI updates in Flutter.

/*
  Sqflite Cars Teaching App
  --------------------------
  This app demonstrates CRUD operations (Create, Read, Update, Delete)
  using the Sqflite package in Flutter.

  Students learn:
  - How to integrate a local SQLite database using Sqflite.
  - How to build a form to insert new data.
  - How to display data in a ListView.
  - How to edit and delete records dynamically.
*/

import 'package:flutter/material.dart';
import 'package:path/path.dart' as p;
import 'package:path_provider/path_provider.dart';
import 'package:sqflite/sqflite.dart';

// -----------------------------
// Main entry point
// -----------------------------
Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized(); // Required before async DB init
  await CarDatabase.instance.database;        // Initialize database
  runApp(const CarApp());
}

// -----------------------------
// Root of the app
// -----------------------------
class CarApp extends StatelessWidget {
  const CarApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Sqflite Car Demo',
      theme: ThemeData(
        colorScheme: ColorScheme.fromSeed(seedColor: Colors.blue),
        useMaterial3: true,
      ),
      home: const CarHomePage(), // Main screen widget
    );
  }
}

// -----------------------------
// Main screen widget
// -----------------------------
class CarHomePage extends StatefulWidget {
  const CarHomePage({super.key});

  @override
  State<CarHomePage> createState() => _CarHomePageState();
}

class _CarHomePageState extends State<CarHomePage> {
  // Text controllers for form input fields
  final _makeController = TextEditingController();
  final _modelController = TextEditingController();
  final _yearController = TextEditingController();
  final _colorController = TextEditingController();

  // List to hold all car records
  List<Car> _cars = <Car>[];

  @override
  void initState() {
    super.initState();
    _refreshCars(); // Load cars from database when app starts
  }

  // Dispose controllers to avoid memory leaks
  @override
  void dispose() {
    _makeController.dispose();
    _modelController.dispose();
    _yearController.dispose();
    _colorController.dispose();
    super.dispose();
  }

  // -----------------------------
  // Helper functions
  // -----------------------------

  // Reloads car list from database
  Future<void> _refreshCars() async {
    final cars = await CarDatabase.instance.fetchCars();
    setState(() {
      _cars = cars;
    });
  }

  // Adds a new car to the database
  Future<void> _addCar() async {
    final make = _makeController.text.trim();
    final model = _modelController.text.trim();
    final color = _colorController.text.trim();
    final year = int.tryParse(_yearController.text.trim());

    // Basic validation
    if (make.isEmpty || model.isEmpty || color.isEmpty || year == null) {
      _showSnackBar('Please provide make, model, year, and color.');
      return;
    }

    await CarDatabase.instance.insertCar(
      Car(make: make, model: model, year: year, color: color),
    );

    // Clear form fields and reload list
    _makeController.clear();
    _modelController.clear();
    _yearController.clear();
    _colorController.clear();

    await _refreshCars();
    _showSnackBar('Car added successfully.');
  }

  // Deletes a car from database
  Future<void> _deleteCar(Car car) async {
    await CarDatabase.instance.deleteCar(car.id!);
    await _refreshCars();
    _showSnackBar('Deleted ${car.displayName}.');
  }

  // Edits an existing car record (opens dialog)
  Future<void> _editCar(Car car) async {
    final edited = await showDialog<Car?>(
      context: context,
      builder: (context) {
        // Prefill text fields with current car data
        final makeController = TextEditingController(text: car.make);
        final modelController = TextEditingController(text: car.model);
        final yearController = TextEditingController(text: car.year.toString());
        final colorController = TextEditingController(text: car.color);

        return AlertDialog(
          title: const Text('Edit Car'),
          content: SingleChildScrollView(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                TextField(controller: makeController, 
                          decoration: const InputDecoration(labelText: 'Make')),
                TextField(controller: modelController, 
                          decoration: const InputDecoration(labelText: 'Model')),
                TextField(controller: yearController, 
                          decoration: const InputDecoration(labelText: 'Year'),
                          keyboardType: TextInputType.number),
                TextField(controller: colorController, 
                          decoration: const InputDecoration(labelText: 'Color')),
              ],
            ),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.of(context).pop(),
              child: const Text('Cancel'),
            ),
            ElevatedButton(
              onPressed: () {
                final make = makeController.text.trim();
                final model = modelController.text.trim();
                final color = colorController.text.trim();
                final year = int.tryParse(yearController.text.trim());

                // Validate edited input
                if (make.isEmpty || model.isEmpty || color.isEmpty || year == null) {
                  Navigator.of(context).pop();
                  _showSnackBar('Invalid input for edit.');
                  return;
                }

                // Create updated car
                final updated = car.copyWith(make: make, model: model, year: year, color: color);
                Navigator.of(context).pop(updated);
              },
              child: const Text('Save'),
            ),
          ],
        );
      },
    );

    // If edited data returned, update database
    if (edited != null) {
      await CarDatabase.instance.updateCar(edited);
      await _refreshCars();
      _showSnackBar('Updated ${edited.displayName}.');
    }
  }

  // Displays a snackbar message
  void _showSnackBar(String message) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(message)));
  }

  // -----------------------------
  // Build UI
  // -----------------------------
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Sqflite Cars Teaching App'),
        backgroundColor: Colors.green,
      ),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          children: [
            // Form card for adding new cars
            Card(
              elevation: 4,
              child: Padding(
                padding: const EdgeInsets.all(16.0),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.stretch,
                  children: [
                    const Text('Add a Car', style: TextStyle(fontSize: 18, 
                                                             fontWeight: ontWeight.bold)),
                    const SizedBox(height: 12),
                    TextField(controller: _makeController, 
                              decoration: const InputDecoration(labelText: 'Make')),
                    TextField(controller: _modelController, 
                              decoration: const InputDecoration(labelText: 'Model')),
                    TextField(controller: _yearController, 
                              decoration: const InputDecoration(labelText: 'Year'),
                              keyboardType: TextInputType.number),
                    TextField(controller: _colorController, 
                              decoration: const InputDecoration(labelText: 'Color')),
                    const SizedBox(height: 16),
                    ElevatedButton.icon(onPressed: _addCar, 
                                        icon: const Icon(Icons.add), 
                                        label: const Text('Add Car')),
                  ],
                ),
              ),
            ),
            const SizedBox(height: 16),
            // ListView for showing all cars
            Expanded(
              child: _cars.isEmpty
                  ? const Center(child: Text('No cars saved yet. Add one above!'))
                  : ListView.separated(
                      itemCount: _cars.length,
                      separatorBuilder: (_, __) => const Divider(height: 1),
                      itemBuilder: (context, index) {
                        final car = _cars[index];
                        return ListTile(
                          title: Text(car.displayName),
                          subtitle: Text('Color: ${car.color}'),
                          trailing: Wrap(
                            spacing: 8,
                            children: [
                              IconButton(onPressed: () => _editCar(car), 
                                         icon: const Icon(Icons.edit), tooltip: 'Edit'),
                              IconButton(onPressed: () => _deleteCar(car), 
                                         icon: const Icon(Icons.delete), 
                                         tooltip: 'Delete'),
                            ],
                          ),
                        );
                      },
                    ),
            ),
          ],
        ),
      ),
    );
  }
}

// -----------------------------
// Car data model
// -----------------------------
class Car {
  const Car({
    this.id,
    required this.make,
    required this.model,
    required this.year,
    required this.color,
  });

  final int? id;
  final String make;
  final String model;
  final int year;
  final String color;

  String get displayName => '$year $make $model';

  // Copy object with optional changes
  Car copyWith({int? id, String? make, String? model, int? year, String? color}) {
    return Car(
      id: id ?? this.id,
      make: make ?? this.make,
      model: model ?? this.model,
      year: year ?? this.year,
      color: color ?? this.color,
    );
  }

  // Convert Car to Map for database storage
  Map<String, Object?> toMap() {
    return {'id': id, 'make': make, 'model': model, 'year': year, 'color': color};
  }

  // Rebuild Car object from Map
  factory Car.fromMap(Map<String, Object?> map) {
    return Car(
      id: map['id'] as int?,
      make: map['make'] as String,
      model: map['model'] as String,
      year: map['year'] as int,
      color: map['color'] as String,
    );
  }
}

// -----------------------------
// Database helper class (Singleton)
// -----------------------------
class CarDatabase {
  CarDatabase._();
  static final CarDatabase instance = CarDatabase._();

  static const _databaseName = 'cars.db';
  static const _databaseVersion = 1;
  Database? _database;

  // Singleton getter for database
  Future<Database> get database async {
    if (_database != null) return _database!;
    _database = await _openDatabase();
    return _database!;
  }

  // Create database and table
  Future<Database> _openDatabase() async {
    final directory = await getApplicationDocumentsDirectory();
    final path = p.join(directory.path, _databaseName);
    return openDatabase(
      path,
      version: _databaseVersion,
      onCreate: (db, version) async {
        await db.execute('''
          CREATE TABLE cars (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            make TEXT NOT NULL,
            model TEXT NOT NULL,
            year INTEGER NOT NULL,
            color TEXT NOT NULL
          )
        ''');
      },
    );
  }

  // -----------------------------
  // CRUD Operations
  // -----------------------------
  Future<List<Car>> fetchCars() async {
    final db = await database;
    final maps = await db.query('cars', orderBy: 'year DESC, make ASC');
    return maps.map(Car.fromMap).toList();
  }

  Future<int> insertCar(Car car) async {
    final db = await database;
    return db.insert('cars', car.toMap()..remove('id'));
  }

  Future<int> updateCar(Car car) async {
    final db = await database;
    return db.update('cars', car.toMap()..remove('id'),
        where: 'id = ?', whereArgs: [car.id]);
  }

  Future<int> deleteCar(int id) async {
    final db = await database;
    return db.delete('cars', where: 'id = ?', whereArgs: [id]);
  }
}
